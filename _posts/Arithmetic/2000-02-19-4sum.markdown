[18. 四数之和](https://leetcode.cn/problems/4sum)

### 题目描述

给你一个由 `n` 个整数组成的数组 `nums`，和一个目标值 `target`。请你找出并返回满足下述全部条件且 **不重复** 的四元组 `[nums[a], nums[b], nums[c], nums[d]]`（若两个四元组元素一一对应，则认为两个四元组重复）：  

- 0 <= a, b, c, d < n  
- a、b、c 和 d 互不相同  
- nums[a] + nums[b] + nums[c] + nums[d] == target  

你可以按 **任意顺序** 返回答案。

---

**示例：**

- 示例 1:
  ```
  输入：nums = [1,0,-1,0,-2,2], target = 0  
  输出：[[-2,-1,1,2],[-2,0,0,2],[-1,0,0,1]]  
  ```

- 示例 2:
  ```
  输入：nums = [2,2,2,2,2], target = 8  
  输出：[[2,2,2,2]]  
  ```

---

**提示：**  

- `1 <= nums.length <= 200`  
- `-10^9 <= nums[i] <= 10^9`  
- `-10^9 <= target <= 10^9`  

---

### 解题思路

这道题要求在数组中找到四个数，使其和等于目标值，并且四元组不能重复。

#### 核心方法：
1. **排序**：  
   - 对数组进行升序排序，以便于后续的双指针查找和去重处理。

2. **遍历**：
   - 使用两个固定指针 `i` 和 `j`，搭配双指针 `left` 和 `right`。
   - 外层两层循环分别确定前两个数字的索引 `i` 和 `j`。
   - 内层双指针 `left` 和 `right` 在剩余数组中寻找满足条件的组合。

3. **双指针查找**：
   - 对于当前固定的前两个数，计算四数之和 `sum`。
   - 如果 `sum == target`，保存该四元组，并移动 `left` 和 `right` 指针，跳过重复元素。
   - 如果 `sum < target`，移动 `left` 指针以增加 `sum`。
   - 如果 `sum > target`，移动 `right` 指针以减小 `sum`。

4. **去重**：
   - 在每次循环中，跳过与前一个数相同的元素，避免产生重复的四元组。

### C语言实现

```c
#include <stdio.h>
#include <stdlib.h>

// 比较函数，用于 qsort 排序
int compare(const void *a, const void *b) {
    return (*(int *)a - *(int *)b);
}

// 四数之和函数
int** fourSum(int* nums, int numsSize, int target, int* returnSize, int** returnColumnSizes) {
    *returnSize = 0;
    if (numsSize < 4) return NULL;

    // 排序数组
    qsort(nums, numsSize, sizeof(int), compare);

    // 分配结果数组的内存
    int maxResults = 1000;
    int** result = (int**)malloc(maxResults * sizeof(int*));
    *returnColumnSizes = (int*)malloc(maxResults * sizeof(int));

    // 遍历数组，寻找四元组
    for (int i = 0; i < numsSize - 3; i++) {
        // 跳过重复元素
        if (i > 0 && nums[i] == nums[i - 1]) continue;

        for (int j = i + 1; j < numsSize - 2; j++) {
            // 跳过重复元素
            if (j > i + 1 && nums[j] == nums[j - 1]) continue;

            int left = j + 1;
            int right = numsSize - 1;

            while (left < right) {
                long sum = (long)nums[i] + nums[j] + nums[left] + nums[right];
                if (sum == target) {
                    // 找到一个四元组，添加到结果中
                    result[*returnSize] = (int*)malloc(4 * sizeof(int));
                    result[*returnSize][0] = nums[i];
                    result[*returnSize][1] = nums[j];
                    result[*returnSize][2] = nums[left];
                    result[*returnSize][3] = nums[right];
                    (*returnColumnSizes)[*returnSize] = 4;
                    (*returnSize)++;

                    // 移动指针，跳过重复元素
                    while (left < right && nums[left] == nums[left + 1]) left++;
                    while (left < right && nums[right] == nums[right - 1]) right--;

                    left++;
                    right--;
                } else if (sum < target) {
                    left++;
                } else {
                    right--;
                }
            }
        }
    }

    return result;
}

// 测试函数
int main() {
    int nums[] = {1, 0, -1, 0, -2, 2};
    int target = 0;
    int returnSize;
    int* returnColumnSizes;
    int** result = fourSum(nums, 6, target, &returnSize, &returnColumnSizes);

    printf("结果:\n");
    for (int i = 0; i < returnSize; i++) {
        printf("[%d, %d, %d, %d]\n", result[i][0], result[i][1], result[i][2], result[i][3]);
        free(result[i]); // 释放内存
    }
    free(result);
    free(returnColumnSizes);

    return 0;
}
```

### C++实现

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

class Solution {
public:
    vector<vector<int>> fourSum(vector<int>& nums, int target) {
        vector<vector<int>> result;
        int n = nums.size();
        if (n < 4) return result;

        // 排序数组
        sort(nums.begin(), nums.end());

        for (int i = 0; i < n - 3; i++) {
            // 跳过重复元素
            if (i > 0 && nums[i] == nums[i - 1]) continue;

            for (int j = i + 1; j < n - 2; j++) {
                // 跳过重复元素
                if (j > i + 1 && nums[j] == nums[j - 1]) continue;

                int left = j + 1, right = n - 1;

                while (left < right) {
                    long sum = (long)nums[i] + nums[j] + nums[left] + nums[right];
                    if (sum == target) {
                        result.push_back({nums[i], nums[j], nums[left], nums[right]});
                        // 跳过重复元素
                        while (left < right && nums[left] == nums[left + 1]) left++;
                        while (left < right && nums[right] == nums[right - 1]) right--;
                        left++;
                        right--;
                    } else if (sum < target) {
                        left++;
                    } else {
                        right--;
                    }
                }
            }
        }

        return result;
    }
};

int main() {
    Solution solution;
    vector<int> nums = {1, 0, -1, 0, -2, 2};
    int target = 0;

    vector<vector<int>> result = solution.fourSum(nums, target);
    cout << "结果:" << endl;
    for (const auto& quad : result) {
        cout << "[" << quad[0] << ", " << quad[1] << ", " << quad[2] << ", " << quad[3] << "]" << endl;
    }

    return 0;
}
```
