[15. 三数之和](https://leetcode.cn/problems/3sum)

### 题目：三数之和 (3Sum)

#### 题目要求：
给你一个整数数组 `nums`，判断 `nums` 中是否存在三个元素 `a，b，c`，使得 `a + b + c = 0`。请你返回所有和为 `0` 的三元组。

注意：
- 答案中不可以包含重复的三元组。

**示例 1:**
```
输入: nums = [-1, 0, 1, 2, -1, -4]
输出: [[-1, -1, 2], [-1, 0, 1]]
```

**示例 2:**
```
输入: nums = []
输出: []
```

**示例 3:**
```
输入: nums = [0]
输出: []
```

**提示：**
- `0 <= nums.length <= 3000`
- `-10^5 <= nums[i] <= 10^5`

---

### 解题思路：

1. **排序与双指针法**：
   - 首先对数组进行排序，这样可以方便地用双指针来寻找和为零的三元组。
   - **排序**：排序后，我们可以轻松地通过确定一个数，然后用双指针查找另外两个数，使得三个数之和为零。
   - **双指针技巧**：对于数组中的每个元素，我们将它作为三元组中的第一个元素，然后利用两个指针来查找另外两个元素。一个指针从当前元素的右侧开始，另一个指针从数组的末尾开始。我们根据三数之和的大小来调整这两个指针的位置：
     - 如果三数之和小于零，左指针右移（增大和）。
     - 如果三数之和大于零，右指针左移（减小和）。
     - 如果三数之和等于零，则找到一个三元组，记录并移动指针以避免重复的三元组。

2. **避免重复**：
   - 通过跳过重复的元素来避免重复的三元组。
   - 在选择第一个数时，我们可以跳过重复的元素，确保每个三元组的第一个元素都是唯一的。

3. **时间复杂度**：
   - 排序需要 `O(n log n)` 时间，双指针搜索需要 `O(n^2)` 时间。因此，整体的时间复杂度为 `O(n^2)`，其中 `n` 是数组的长度。

#### C 语言解法：

```c
#include <stdio.h>
#include <stdlib.h>

#define MAX_RESULT_SIZE 1000

// 辅助函数：比较函数，用于排序
int compare(const void* a, const void* b) {
    return (*(int*)a - *(int*)b);
}

// 函数：查找所有和为0的三元组
int** threeSum(int* nums, int numsSize, int* returnSize, int** returnColumnSizes) {
    // 如果数组长度小于3，返回空数组
    if (numsSize < 3) {
        *returnSize = 0;
        return NULL;
    }

    // 排序数组
    qsort(nums, numsSize, sizeof(int), compare);

    int** result = (int**)malloc(MAX_RESULT_SIZE * sizeof(int*)); // 存储结果
    *returnColumnSizes = (int*)malloc(MAX_RESULT_SIZE * sizeof(int)); // 存储每个三元组的大小
    *returnSize = 0;

    for (int i = 0; i < numsSize - 2; i++) {
        // 跳过重复元素，确保第一个数是唯一的
        if (i > 0 && nums[i] == nums[i - 1]) {
            continue;
        }

        int left = i + 1, right = numsSize - 1;
        while (left < right) {
            int sum = nums[i] + nums[left] + nums[right];
            if (sum == 0) {
                // 找到一个三元组
                result[*returnSize] = (int*)malloc(3 * sizeof(int));
                result[*returnSize][0] = nums[i];
                result[*returnSize][1] = nums[left];
                result[*returnSize][2] = nums[right];
                (*returnColumnSizes)[*returnSize] = 3;
                (*returnSize)++;

                // 跳过重复元素
                while (left < right && nums[left] == nums[left + 1]) left++;
                while (left < right && nums[right] == nums[right - 1]) right--;
                left++;
                right--;
            } else if (sum < 0) {
                left++;
            } else {
                right--;
            }
        }
    }

    return result;
}

int main() {
    int nums[] = {-1, 0, 1, 2, -1, -4};
    int numsSize = sizeof(nums) / sizeof(nums[0]);
    int returnSize = 0;
    int* returnColumnSizes = NULL;

    int** result = threeSum(nums, numsSize, &returnSize, &returnColumnSizes);

    // 输出结果
    for (int i = 0; i < returnSize; i++) {
        printf("[");
        for (int j = 0; j < 3; j++) {
            printf("%d", result[i][j]);
            if (j < 2) printf(", ");
        }
        printf("]\n");
    }

    return 0;
}
```

#### C++ 解法：

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        vector<vector<int>> result;
        int n = nums.size();

        if (n < 3) return result;  // 如果数组长度小于3，返回空

        sort(nums.begin(), nums.end());  // 排序数组

        for (int i = 0; i < n - 2; i++) {
            // 跳过重复元素，确保第一个数是唯一的
            if (i > 0 && nums[i] == nums[i - 1]) continue;

            int left = i + 1, right = n - 1;
            while (left < right) {
                int sum = nums[i] + nums[left] + nums[right];
                if (sum == 0) {
                    result.push_back({nums[i], nums[left], nums[right]});
                    // 跳过重复元素
                    while (left < right && nums[left] == nums[left + 1]) left++;
                    while (left < right && nums[right] == nums[right - 1]) right--;
                    left++;
                    right--;
                } else if (sum < 0) {
                    left++;
                } else {
                    right--;
                }
            }
        }

        return result;
    }
};

int main() {
    Solution solution;
    vector<int> nums = {-1, 0, 1, 2, -1, -4};
    vector<vector<int>> result = solution.threeSum(nums);

    // 输出结果
    for (const auto& triplet : result) {
        cout << "[";
        for (int i = 0; i < triplet.size(); i++) {
            cout << triplet[i];
            if (i < triplet.size() - 1) cout << ", ";
        }
        cout << "]" << endl;
    }

    return 0;
}
```

### 代码解释：

1. **C 语言实现**：
   - 首先对数组进行排序。
   - 遍历每个元素，并使用两个指针分别指向当前元素的后面和数组的末尾。
   - 通过调整两个指针的值，寻找满足条件的三元组。
   - 为了避免重复的三元组，跳过相同的元素。

2. **C++ 实现**：
   - 使用 `vector<vector<int>>` 存储结果，动态管理内存。
   - 排序并使用双指针法查找三元组。
   - 在添加三元组后，跳过相同的元素以避免重复。

### 核心算法：
- **双指针法**：通过固定一个数，然后用两个指针查找剩余两个数。通过调整指针位置来减少搜索空间，优化查找过程。
