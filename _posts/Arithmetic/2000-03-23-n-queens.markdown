[51. N 皇后](https://leetcode.cn/problems/n-queens)

### 题目描述

n 皇后问题研究的是如何将 n 个皇后放置在 n×n 的棋盘上，并且使皇后彼此之间不能相互攻击。

给你一个整数 `n` ，返回所有不同的 n 皇后问题的解决方案。

每一种解法包含一个不同的 n 皇后问题的棋子放置方案，该方案中 `'Q'` 和 `'.'` 分别表示皇后和空位。

---

### 示例：

#### 示例 1：
```
输入：n = 4
输出：[[".Q..","...Q","Q...","..Q."],
      ["..Q.","Q...","...Q",".Q.."]]
解释：如上图所示，4 皇后问题存在两个不同的解法。
```

#### 示例 2：
```
输入：n = 1
输出：[["Q"]]
```

---

### 提示：
- `1 <= n <= 9`

---

### 解题思路

1. **回溯算法**：
   - 使用回溯算法在棋盘中逐行放置皇后，确保在放置每一个皇后时，不会与已经放置的皇后冲突。

2. **冲突检查**：
   - 冲突条件：
     - 同列冲突：当前列已有皇后。
     - 主对角线冲突：`row - col` 值相等。
     - 副对角线冲突：`row + col` 值相等。
   - 使用三个集合（或数组）记录被占用的列和对角线。

3. **状态恢复**：
   - 放置一个皇后后，进入下一行递归处理。
   - 如果当前路径无法放置皇后，则回溯并恢复棋盘状态。

4. **结果存储**：
   - 当成功放置了 n 个皇后（即当前行索引等于 n），将棋盘状态记录为一个解。

### C 语言实现

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// 定义结果存储结构
typedef struct {
    char*** solutions; // 存储每个解法的二维数组
    int* columnSizes;  // 每个解法的行大小
    int size;          // 当前解法数量
    int capacity;      // 总分配容量
} Result;

// 初始化结果结构
Result* initResult(int capacity) {
    Result* res = (Result*)malloc(sizeof(Result));
    res->solutions = (char***)malloc(sizeof(char**) * capacity);
    res->columnSizes = (int*)malloc(sizeof(int) * capacity);
    res->size = 0;
    res->capacity = capacity;
    return res;
}

// 添加解法到结果中
void addSolution(Result* res, char** board, int n) {
    if (res->size == res->capacity) {
        res->capacity *= 2;
        res->solutions = (char***)realloc(res->solutions, sizeof(char**) * res->capacity);
        res->columnSizes = (int*)realloc(res->columnSizes, sizeof(int) * res->capacity);
    }
    char** solution = (char**)malloc(sizeof(char*) * n);
    for (int i = 0; i < n; i++) {
        solution[i] = strdup(board[i]);
    }
    res->solutions[res->size] = solution;
    res->columnSizes[res->size] = n;
    res->size++;
}

// 判断当前位置是否可以放置皇后
int isValid(char** board, int row, int col, int n) {
    for (int i = 0; i < row; i++) {
        if (board[i][col] == 'Q') return 0; // 同列冲突
    }
    for (int i = row - 1, j = col - 1; i >= 0 && j >= 0; i--, j--) {
        if (board[i][j] == 'Q') return 0; // 主对角线冲突
    }
    for (int i = row - 1, j = col + 1; i >= 0 && j < n; i--, j++) {
        if (board[i][j] == 'Q') return 0; // 副对角线冲突
    }
    return 1;
}

// 回溯求解 n 皇后问题
void backtrack(char** board, int row, int n, Result* res) {
    if (row == n) {
        addSolution(res, board, n);
        return;
    }
    for (int col = 0; col < n; col++) {
        if (isValid(board, row, col, n)) {
            board[row][col] = 'Q';
            backtrack(board, row + 1, n, res);
            board[row][col] = '.'; // 恢复状态
        }
    }
}

// 主函数
char*** solveNQueens(int n, int* returnSize, int** columnSizes) {
    Result* res = initResult(100);
    char** board = (char**)malloc(sizeof(char*) * n);
    for (int i = 0; i < n; i++) {
        board[i] = (char*)malloc(sizeof(char) * (n + 1));
        memset(board[i], '.', n);
        board[i][n] = '\0';
    }

    backtrack(board, 0, n, res);

    for (int i = 0; i < n; i++) {
        free(board[i]);
    }
    free(board);

    *returnSize = res->size;
    *columnSizes = res->columnSizes;
    return res->solutions;
}

// 测试函数
int main() {
    int n = 4;
    int returnSize;
    int* columnSizes;
    char*** solutions = solveNQueens(n, &returnSize, &columnSizes);

    printf("Number of solutions: %d\n", returnSize);
    for (int i = 0; i < returnSize; i++) {
        printf("Solution %d:\n", i + 1);
        for (int j = 0; j < columnSizes[i]; j++) {
            printf("%s\n", solutions[i][j]);
            free(solutions[i][j]);
        }
        free(solutions[i]);
    }
    free(solutions);
    free(columnSizes);

    return 0;
}
```

### C++ 语言实现

```cpp
#include <iostream>
#include <vector>
#include <string>

using namespace std;

class Solution {
public:
    vector<vector<string>> solveNQueens(int n) {
        vector<vector<string>> results;
        vector<string> board(n, string(n, '.'));
        vector<int> cols(n, 0), diag1(2 * n - 1, 0), diag2(2 * n - 1, 0);

        backtrack(results, board, cols, diag1, diag2, 0, n);
        return results;
    }

private:
    void backtrack(vector<vector<string>>& results, vector<string>& board,
                   vector<int>& cols, vector<int>& diag1, vector<int>& diag2, int row, int n) {
        if (row == n) {
            results.push_back(board);
            return;
        }
        for (int col = 0; col < n; col++) {
            if (cols[col] || diag1[row - col + n - 1] || diag2[row + col]) continue;

            board[row][col] = 'Q';
            cols[col] = diag1[row - col + n - 1] = diag2[row + col] = 1;

            backtrack(results, board, cols, diag1, diag2, row + 1, n);

            board[row][col] = '.';
            cols[col] = diag1[row - col + n - 1] = diag2[row + col] = 0;
        }
    }
};

// 测试函数
int main() {
    Solution sol;
    int n = 4;
    vector<vector<string>> solutions = sol.solveNQueens(n);

    cout << "Number of solutions: " << solutions.size() << endl;
    for (int i = 0; i < solutions.size(); i++) {
        cout << "Solution " << i + 1 << ":" << endl;
        for (const string& row : solutions[i]) {
            cout << row << endl;
        }
    }
    return 0;
}
```
