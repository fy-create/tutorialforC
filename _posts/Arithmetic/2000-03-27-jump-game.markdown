[55. 跳跃游戏](https://leetcode.cn/problems/jump-game)

### 题目描述：
给定一个非负整数数组 `nums` ，你最初位于数组的第一个下标。

数组中的每个元素代表你在该位置可以跳跃的最大长度。

判断你是否能够到达最后一个下标。

**示例：**

**输入：** nums = [2,3,1,1,4]

**输出：** true

**解释：** 我们可以先跳 1 步，从下标 0 到达下标 1，然后再从下标 1 跳 3 步到达最后一个下标。

**输入：** nums = [3,2,1,0,4]

**输出：** false

**解释：** 无论怎样，总会到达下标 3 时跳跃步数为 0 ，所以永远无法到达最后一个下标。

**提示：**
- 1 <= nums.length <= 10^4
- 0 <= nums[i] <= 10^5

### 解题思路：

1. **贪心算法：**
   - 从左到右遍历数组，维护一个变量 `maxReach` 表示当前可以到达的最远位置。
   - 对于当前位置 `i`，如果 `i > maxReach`，说明无法到达位置 `i`，返回 `false`。
   - 否则，更新 `maxReach = max(maxReach, i + nums[i])`。
   - 如果遍历结束时 `maxReach >= nums.length - 1`，说明可以到达最后一个位置，返回 `true`。

2. **时间复杂度：**
   - 时间复杂度为 O(n)，其中 n 是数组的长度。每个元素最多访问一次。

3. **空间复杂度：**
   - 空间复杂度为 O(1)，只使用了常数个额外变量。

```c
#include <stdio.h>
#include <stdbool.h>

bool canJump(int* nums, int numsSize) {
    int maxReach = 0; // 当前能够到达的最远位置
    for (int i = 0; i < numsSize; i++) {
        if (i > maxReach) {
            return false; // 无法到达当前位置
        }
        if (maxReach >= numsSize - 1) {
            return true; // 已经可以到达最后一个位置
        }
        maxReach = (i + nums[i] > maxReach) ? i + nums[i] : maxReach; // 更新最远位置
    }
    return false;
}

int main() {
    int nums1[] = {2, 3, 1, 1, 4};
    int nums2[] = {3, 2, 1, 0, 4};

    printf("Can jump (example 1): %s\n", canJump(nums1, 5) ? "true" : "false");
    printf("Can jump (example 2): %s\n", canJump(nums2, 5) ? "true" : "false");

    return 0;
}
```

```cpp
#include <iostream>
#include <vector>
using namespace std;

class Solution {
public:
    bool canJump(vector<int>& nums) {
        int maxReach = 0; // 当前能够到达的最远位置
        for (int i = 0; i < nums.size(); i++) {
            if (i > maxReach) {
                return false; // 无法到达当前位置
            }
            if (maxReach >= nums.size() - 1) {
                return true; // 已经可以到达最后一个位置
            }
            maxReach = max(maxReach, i + nums[i]); // 更新最远位置
        }
        return false;
    }
};

int main() {
    Solution sol;
    vector<int> nums1 = {2, 3, 1, 1, 4};
    vector<int> nums2 = {3, 2, 1, 0, 4};

    cout << "Can jump (example 1): " << (sol.canJump(nums1) ? "true" : "false") << endl;
    cout << "Can jump (example 2): " << (sol.canJump(nums2) ? "true" : "false") << endl;

    return 0;
}
```
