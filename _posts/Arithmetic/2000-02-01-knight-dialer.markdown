### 题目要求
国际象棋中的骑士可以按“L”形移动到8个可能的位置之一，如下图所示：  

```
1 2 3  
4 5 6  
7 8 9  
  0  
```

骑士的行动路径如下：
- 从数字1可以跳到数字6和8；
- 从数字2可以跳到数字7和9；
- 从数字3可以跳到数字4和8；
- 从数字4可以跳到数字3、9和0；
- 从数字5无法移动；
- 从数字6可以跳到数字1、7和0；
- 从数字7可以跳到数字2和6；
- 从数字8可以跳到数字1和3；
- 从数字9可以跳到数字2和4；
- 从数字0可以跳到数字4和6。

要求：
- 给定一个整数`n`，计算可以拨出长度为`n`的不同电话号码的数量。
- 结果需对`10^9 + 7`取模。

**示例：**  
```
输入：n = 1  
输出：10  
解释：我们需要拨一个长度为1的数字，因此将骑士放在键盘上的任何数字上均满足条件。  

输入：n = 2  
输出：20  
解释：长度为2的有效号码有：[04, 06, 16, 18, 27, 29, 34, 38, 40, 43, 49, 60, 61, 67, 72, 76, 81, 83, 92, 94]。  
```

**提示：**  
- 1 <= n <= 5000  

---

### 解题思路
我们使用动态规划来解决这个问题。  
#### 核心思想：
1. 定义`dp[i][j]`表示经过`i`次跳跃后，以数字`j`结尾的电话号码的数量。  
2. 初始时，长度为1的电话号码只有一个，即数字本身：`dp[0][j] = 1`。  
3. 根据骑士的跳跃规则，从上一个状态`dp[i-1][k]`转移到当前状态`dp[i][j]`。  
4. 最终结果是所有`dp[n-1][j]`的累加和。

#### 动态规划更新公式：
$$ dp[ i ][ j ] = \sum_{k \in moves[ j ] } dp[ i-1 ][ k ] $$

#### 优化：
- 由于结果可能很大，在每次计算中需要对`10^9 + 7`取模。
- 可以用滚动数组优化空间复杂度，从`O(n×10)`降至`O(10)`。

### C语言实现
```c
#include <stdio.h>

#define MOD 1000000007

// 定义骑士从每个数字可以跳到的数字
const int moves[10][3] = {
    {4, 6, -1},    // 0 -> 4, 6
    {6, 8, -1},    // 1 -> 6, 8
    {7, 9, -1},    // 2 -> 7, 9
    {4, 8, -1},    // 3 -> 4, 8
    {0, 3, 9},     // 4 -> 0, 3, 9
    {-1, -1, -1},  // 5 -> 无法跳跃到其他数字
    {0, 1, 7},     // 6 -> 0, 1, 7
    {2, 6, -1},    // 7 -> 2, 6
    {1, 3, -1},    // 8 -> 1, 3
    {2, 4, -1}     // 9 -> 2, 4
};

// 计算长度为 n 的不同电话号码的数量
int knightDialer(int n) {
    int dp[2][10] = {0};
    // 初始化长度为 1 的电话号码
    for (int j = 0; j < 10; j++) {
        dp[0][j] = 1;
    }
    // 使用滚动数组计算
    for (int i = 1; i < n; i++) {
        for (int j = 0; j < 10; j++) {
            dp[i % 2][j] = 0;  // 清空当前行
            for (int k = 0; k < 3; k++) {
                int prev = moves[j][k];
                if (prev != -1) {
                    dp[i % 2][j] = (dp[i % 2][j] + dp[(i - 1) % 2][prev]) % MOD;
                }
            }
        }
    }
    // 计算总数
    int result = 0;
    for (int j = 0; j < 10; j++) {
        result = (result + dp[(n - 1) % 2][j]) % MOD;
    }
    return result;
}

// 测试函数
int main() {
    int n = 2;
    printf("长度为 %d 的不同电话号码的数量: %d\n", n, knightDialer(n));
    return 0;
}
```

### C++实现
```cpp
#include <iostream>
#include <vector>
using namespace std;

class Solution {
public:
    int knightDialer(int n) {
        const int MOD = 1e9 + 7;
        vector<vector<int>> moves = {
            {4, 6},    // 0 -> 4, 6
            {6, 8},    // 1 -> 6, 8
            {7, 9},    // 2 -> 7, 9
            {4, 8},    // 3 -> 4, 8
            {0, 3, 9}, // 4 -> 0, 3, 9
            {},        // 5 -> 无法跳跃到其他数字
            {0, 1, 7}, // 6 -> 0, 1, 7
            {2, 6},    // 7 -> 2, 6
            {1, 3},    // 8 -> 1, 3
            {2, 4}     // 9 -> 2, 4
        };
        vector<vector<int>> dp(2, vector<int>(10, 0));
        // 初始化长度为 1 的电话号码
        fill(dp[0].begin(), dp[0].end(), 1);
        // 动态规划
        for (int i = 1; i < n; ++i) {
            fill(dp[i % 2].begin(), dp[i % 2].end(), 0);
            for (int j = 0; j < 10; ++j) {
                for (int prev : moves[j]) {
                    dp[i % 2][j] = (dp[i % 2][j] + dp[(i - 1) % 2][prev]) % MOD;
                }
            }
        }
        // 计算总数
        int result = 0;
        for (int j = 0; j < 10; ++j) {
            result = (result + dp[(n - 1) % 2][j]) % MOD;
        }
        return result;
    }
};

int main() {
    Solution solution;
    int n = 2;
    cout << "长度为 " << n << " 的不同电话号码的数量: " << solution.knightDialer(n) << endl;
    return 0;
}
```
